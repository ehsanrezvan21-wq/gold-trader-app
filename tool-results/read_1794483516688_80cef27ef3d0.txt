    53→function todayKey() {
    54→  const d = new Date()
    55→  return `${d.getFullYear()}-${d.getMonth() + 1}-${d.getDate()}`

    58→function loadState(): PersistState {
    59→  try {
    60→    if (fs.existsSync(STATE_FILE)) {
    61→      const s = JSON.parse(fs.readFileSync(STATE_FILE, 'utf8')) as PersistState
    62→      return {
    63→        settings: { ...DEFAULT_SETTINGS, ...s.settings, botRunning: false },
    64→        balance: typeof s.balance === 'number' ? s.balance : 100,
    65→        dayPnl: s.dayPnl ?? 0,
    66→        dayKey: s.dayKey ?? todayKey(),
    67→        // merge for migration: older state.json files lack the new stat fields
    68→        stats: { ...freshStats(), ...(s.stats ?? {}) },
    69→        trades: Array.isArray(s.trades) ? s.trades.slice(0, MAX_HISTORY) : [],
    70→      }
    71→    }
    72→  } catch { /* corrupted state -> reset */ }
    73→  return {
    74→    settings: { ...DEFAULT_SETTINGS },
    75→    balance: 100,
    76→    dayPnl: 0,
    77→    dayKey: todayKey(),
    78→    stats: freshStats(),
    79→    trades: [],
    80→  }
    81→}

   239→  /** Manual BUY/SELL from the user, bypassing the AI confidence gate and the post-loss cooldown. */
   240→  openManualTrade(direction: 'BUY' | 'SELL', symbol?: string): { ok: boolean; reason?: string } {
   241→    const sym = symbolConfig(symbol || this.settings.activeSymbol).symbol
   242→    const check = this.canOpen()
   243→    if (!check.ok) {
   244→      this.log('warn', `معامله دستی رد شد — ${check.reason}`)
   245→      this.onEmit()
   246→      return { ok: false, reason: check.reason }
   247→    }
   248→    const r = this.tryOpenTrade(direction, 100, 'معامله دستی کاربر', 'MANUAL', true, sym)
   249→    if (r.ok) {
   250→      this.log('success', `معامله دستی ${direction === 'BUY' ? 'خرید' : 'فروش'} ${symbolConfig(sym).fa} باز شد — بستن خودکار در ${this.settings.tradeDurationSec} ثانیه`)
   251→    }
   252→    this.onEmit()
   253→    return { ok: r.ok, reason: r.ok ? undefined : r.reason }
   254→  }

   257→  /** Effective stake for the next demo trade: fixed $ or % of balance (auto sizing). */
   258→  private currentStake(): number {
   259→    const pct = this.settings.autoStakePct
   260→    if (pct > 0) {
   261→      const auto = +(this.balance * pct / 100).toFixed(2)
   262→      return Math.min(100, Math.max(1, auto))
   263→    }
   264→    return this.settings.stake
   265→  }

   267→  /** Win-rate (0..1) of this symbol's last closed WIN/LOSS trades — undefined if no data. */
   268→  private perSymbolAccuracy(symbol: string): number | undefined {
   269→    const rows = this.trades
   270→      .filter(t => t.symbol === symbol && t.status === 'closed' && (t.result === 'WIN' || t.result === 'LOSS'))
   271→      .slice(0, 10)
   272→    if (rows.length < 3) return undefined
   273→    const wins = rows.filter(t => t.result === 'WIN').length
   274→    return wins / rows.length
   275→  }
   276→
   277→  /** Last-10-trades summary of the whole account, for the GLM prompt. */
   278→  private recentSummary(): RecentContext {
   279→    const rows = this.trades
   280→      .filter(t => t.status === 'closed' && (t.result === 'WIN' || t.result === 'LOSS'))
   281→      .slice(0, 10)
   282→    const wins = rows.filter(t => t.result === 'WIN').length
   283→    return { wins, losses: rows.length - wins, streak: this.stats.streak }
   284→  }

   347→  private onTick() {
   348→    if (this.dayKey !== todayKey()) {
   349→      this.resetDay()
   350→      this.log('info', 'روز معاملاتی جدید — آمار روزانه صفر شد')
   351→    }
   352→    this.emitTicks()
   353→
   354→    // settle due turbo trades (each with its own symbol's price)
   355→    const now = Date.now()
   356→    for (const trade of this.trades.filter(t => t.status === 'open')) {
   357→      if (trade.mode === 'demo' && now >= trade.closeTime) {
   358→        this.settleDemo(trade, this.priceOf(trade.symbol))
   359→      } else if (trade.mode === 'mt5' && trade.positionId && now >= trade.closeTime) {
   360→        this.settleMt5(trade, false)
   361→      }
   362→    }
   363→
   364→    this.onEmit()
   365→  }

   367→  private async aiCycle() {
   368→    if (this.aiBusy) return
   369→    // technical analysis always runs at full turbo cadence; only the GLM call
   370→    // is throttled separately by its own rate-limit backoff
   371→    if (Date.now() - this.lastSignalAt < this.settings.signalIntervalSec * 1000) return
   372→    this.aiBusy = true
   373→    try {
   374→      const activeSym = this.settings.activeSymbol
   375→
   376→      // 1) GLM deep analysis of the active symbol (rate-limit aware)
   377→      let glmRan = false
   378→      const glmReady = Date.now() - this.lastGlmTryAt >= this.rateBackoff * 1000
   379→      if (glmReady) {
   380→        this.lastGlmTryAt = Date.now()
   381→        try {
   382→          const candles = this.markets[activeSym].b1m.closed.slice(-40)
   383→          const verdict = await analyzeMarket(candles, activeSym, this.recentSummary())
   384→          this.rateBackoff = 0
   385→          // If GLM hesitates (HOLD) but technicals clearly lean one way, follow the technical signal
   386→          let final = verdict
   387→          if (verdict.action === 'HOLD') {
   388→            const local = localVerdict(candles)
   389→            if (local.action !== 'HOLD' && local.confidence >= 60) {
   390→              final = { ...local, reason: `${local.reason} — تأیید تکنیکال` }
   391→            }
   392→          }
   393→          this.applyVerdict(final, 'AI', activeSym)
   394→          glmRan = true
   395→        } catch (e: any) {
   396→          const msg = String(e?.message || 'نامشخص')
   397→          const rateLimited = msg.includes('429') || msg.toLowerCase().includes('too many')
   398→          if (rateLimited) {
   399→            // exponential backoff up to 120s — for the GLM call only
   400→            this.rateBackoff = Math.min((this.rateBackoff || 15) * 1.6, 120)
   401→          }
   402→          this.log('info', `AI در دسترس نیست (${rateLimited ? 'محدودیت نرخ' : 'خطا'}) — ادامه با تحلیل تکنیکال`)
   403→        }
   404→      }
   405→
   406→      // 2) technical scan of the other symbols so the bot never misses opportunities
   407→      for (const sym of SYMBOLS) {
   408→        if (sym === activeSym && glmRan) continue
   409→        try {
   410→          const candles = this.markets[sym].b1m.closed.slice(-40)
   411→          const verdict = localVerdict(candles, this.perSymbolAccuracy(sym))
   412→          this.applyVerdict(verdict, 'تکنیکال', sym)
   413→        } catch (e: any) {
   414→          this.log('error', `خطای تحلیل تکنیکال ${symbolConfig(sym).fa}: ${e?.message || 'نامشخص'}`)
   415→        }
   416→      }
   417→    } finally {
   418→      this.aiBusy = false
   419→      this.lastSignalAt = Date.now()
   420→      this.onEmit()
   421→    }
   422→  }

   424→  private applyVerdict(
   425→    verdict: { action: 'BUY' | 'SELL' | 'HOLD'; confidence: number; reason: string; indicators: ReturnType<typeof computeIndicators> },
   426→    source: 'AI' | 'تکنیکال',
   427→    symbol: string,
   428→  ) {
   429→    const cfg = symbolConfig(symbol)
   430→    const price = this.priceOf(symbol)
   431→
   432→    let acted = false
   433→    let actedTrade: Trade | null = null
   434→    let blockedReason: string | undefined
   435→    if (verdict.action !== 'HOLD' && verdict.confidence >= this.settings.minConfidence && this.settings.botRunning) {
   436→      const r = this.tryOpenTrade(verdict.action as 'BUY' | 'SELL', verdict.confidence, verdict.reason, undefined, false, symbol)
   437→      acted = r.ok
   438→      actedTrade = r.trade ?? null
   439→      blockedReason = r.reason
   440→    }
   441→
   442→    const signal: AISignal = {
   443→      id: Math.random().toString(36).slice(2),
   444→      time: Date.now(),
   445→      symbol,
   446→      action: verdict.action,
   447→      confidence: verdict.confidence,
   448→      reason: verdict.reason,
   449→      price,
   450→      indicators: verdict.indicators,
   451→      acted,
   452→    }
   453→    // keep signal history: prefer mixing symbols but recent-first ordering stays
   454→    this.signals.unshift(signal)
   455→    if (this.signals.length > 30) this.signals.pop()
   456→
   457→    const label = source === 'AI' ? 'AI' : 'تحلیل تکنیکال (AI موقتاً در دسترس نیست)'
   458→    const tag = verdict.action === 'HOLD' ? '' : acted
   459→      ? ' → معامله باز شد'
   460→      : verdict.confidence >= this.settings.minConfidence
   461→        ? ` (${blockedReason || 'بات غیرفعال یا ظرفیت تکمیل'})`
   462→        : ` (زیر حد اطمینان ${this.settings.minConfidence}%)`
   463→    this.log(source === 'AI' ? 'ai' : 'info', `${label} [${cfg.fa}]: ${verdict.action === 'BUY' ? 'خرید' : verdict.action === 'SELL' ? 'فروش' : 'انتظار'} — اطمینان ${verdict.confidence}٪ — ${verdict.reason}${tag}`)
   464→    if (actedTrade) this.log('success', `معامله ${verdict.action === 'BUY' ? 'خرید' : 'فروش'} ${cfg.fa} با حجم $${actedTrade.stake} باز شد — هدف: ${actedTrade.durationSec} ثانیه`)
   465→  }

   467→  private canOpen(includeCooldown = false): { ok: boolean; reason?: string } {
   468→    if (this.pausedReason) return { ok: false, reason: this.pausedReason }
   469→    if (this.dayPnl <= -this.settings.dailyLossCap) {
   470→      this.pausedReason = `سقف ضرر روزانه (${this.settings.dailyLossCap}$) لمس شد — بات متوقف است`
   471→      this.settings.botRunning = false
   472→      this.log('warn', this.pausedReason)
   473→      return { ok: false, reason: this.pausedReason }
   474→    }
   475→    // post-loss cooldown: prevents revenge-trading cascades right after a loss
   476→    if (includeCooldown && this.lastLossAt > 0) {
   477→      const waitMs = this.settings.cooldownAfterLossSec * 1000
   478→      const elapsed = Date.now() - this.lastLossAt
   479→      if (elapsed < waitMs) {
   480→        return { ok: false, reason: `استراحت بعد از باخت — ${Math.ceil((waitMs - elapsed) / 1000)} ثانیه تا معامله بعدی` }
   481→      }
   482→    }
   483→    const openCount = this.trades.filter(t => t.status === 'open').length
   484→    if (openCount >= this.settings.maxConcurrent) return { ok: false, reason: 'ظرفیت معاملات همزمان تکمیل است' }
   485→    if (this.settings.mode === 'demo' && this.balance < this.currentStake()) {
   486→      this.pausedReason = 'موجودی کافی نیست'
   487→      return { ok: false, reason: this.pausedReason }
   488→    }
   489→    return { ok: true }
   490→  }

   492→  private tryOpenTrade(direction: 'BUY' | 'SELL', confidence: number, reason: string, comment?: string, skipCooldown = false, symbol = this.settings.activeSymbol): { ok: boolean; reason?: string; trade?: Trade } {
   493→    const check = this.canOpen(!skipCooldown)
   494→    if (!check.ok) return { ok: false, reason: check.reason }
   495→    const now = Date.now()
   496→    const base: Omit<Trade, 'openPrice' | 'mode' | 'positionId' | 'lots' | 'stake'> = {
   497→      id: Math.random().toString(36).slice(2),
   498→      symbol,
   499→      direction,
   500→      openTime: now,
   501→      closeTime: now + this.settings.tradeDurationSec * 1000,
   502→      durationSec: this.settings.tradeDurationSec,
   503→      status: 'open',
   504→      confidence,
   505→      aiReason: reason,
   506→      comment: comment ?? `AI ${confidence}%`,
   507→    }
   508→
   509→    if (this.settings.mode === 'demo') {
   510→      const stake = this.currentStake()
   511→      if (stake > this.balance) return { ok: false, reason: 'موجودی کافی نیست' }
   512→      const trade: Trade = { ...base, mode: 'demo', stake, lots: 0, openPrice: this.priceOf(symbol) }
   513→      this.trades.unshift(trade)
   514→      if (this.trades.length > MAX_HISTORY) this.trades.pop()
   515→      this.save()
   516→      return { ok: true, trade }
   517→    }
   518→
   519→    // MT5 real mode
   520→    const trade: Trade = { ...base, mode: 'mt5', stake: 0, lots: this.settings.mt5Lots, openPrice: this.priceOf(symbol) }
   521→    // optimistic insert; confirm with API
   522→    this.trades.unshift(trade)
   523→    if (this.trades.length > MAX_HISTORY) this.trades.pop()
   524→    mt5Open(this.settings, direction, base.comment, symbol)
   525→      .then(res => {
   526→        if (res.ok) {
   527→          trade.positionId = res.positionId
   528→          trade.openPrice = res.openPrice ?? trade.openPrice
   529→          this.log('success', `سفارش واقعی ${direction} روی MT5 ثبت شد (${symbolConfig(symbol).fa}، حجم ${this.settings.mt5Lots} لات)`)
   530→        } else {
   531→          trade.status = 'closed'
   532→          trade.result = 'DRAW'
   533→          trade.pnl = 0
   534→          this.log('error', res.error || 'ثبت سفارش MT5 ناموفق بود')
   535→        }
   536→        this.save()
   537→        this.onEmit()
   538→      })
   539→    return { ok: true, trade }
   540→  }

   542→  private settleDemo(trade: Trade, closePrice: number) {
   543→    trade.status = 'closed'
   544→    trade.closePrice = +closePrice.toFixed(symbolConfig(trade.symbol).decimals)
   545→    const result = demoSettle(trade.direction, trade.openPrice, trade.closePrice)
   546→    trade.result = result
   547→    trade.pnl = result === 'WIN' ? +(trade.stake * this.settings.payout).toFixed(2)
   548→      : result === 'LOSS' ? -trade.stake : 0
   549→    this.balance = +(this.balance + (trade.pnl ?? 0)).toFixed(2)
   550→    this.applyTradeResult(trade)
   551→    this.log(result === 'WIN' ? 'success' : 'warn',
   552→      `معامله ${result === 'WIN' ? 'برد ✚' : 'باخت ✖'} — ${trade.direction} ${symbolConfig(trade.symbol).fa} — سود/زیان: ${trade.pnl >= 0 ? '+' : ''}$${trade.pnl} (بسته شده در ${trade.durationSec} ثانیه)`)
   553→    if (result === 'LOSS' && this.settings.cooldownAfterLossSec > 0) {
   554→      this.log('info', `استراحت ${this.settings.cooldownAfterLossSec} ثانیه‌ای بعد از باخت فعال شد`)
   555→    }
   556→  }

   558→  private async settleMt5(trade: Trade, manual: boolean) {
   559→    if (!trade.positionId) return
   560→    const res = await mt5Close(this.settings, trade.positionId)
   561→    if (!res.ok) {
   562→      this.log('error', res.error || 'بستن پوزیشن ناموفق بود')
   563→      return
   564→    }
   565→    // fetch actual profit from positions history via account info refresh
   566→    trade.status = 'closed'
   567→    trade.closeTime = Date.now()
   568→    const acct = await mt5Account(this.settings)
   569→    const pnl = typeof res.pnl === 'number' ? res.pnl : 0
   570→    trade.pnl = +pnl.toFixed(2)
   571→    trade.result = pnl > 0 ? 'WIN' : pnl < 0 ? 'LOSS' : 'DRAW'
   572→    if (acct.ok && acct.info) {
   573→      this.mt5 = acct.info
   574→      this.balance = acct.info.balance
   575→    } else {
   576→      this.balance = +(this.balance + pnl).toFixed(2)
   577→    }
   578→    this.applyTradeResult(trade)
   579→    this.log(pnl >= 0 ? 'success' : 'warn', `پوزیشن MT5 ${manual ? 'دستی ' : ''}بسته شد — سود/زیان: ${pnl >= 0 ? '+' : ''}$${trade.pnl}`)
   580→  }

   582→  private applyTradeResult(trade: Trade) {
   583→    this.stats.totalTrades++
   584→    this.stats.totalPnl = +(this.stats.totalPnl + (trade.pnl ?? 0)).toFixed(2)
   585→    this.stats.todayTrades++
   586→    this.stats.todayPnl = +(this.stats.todayPnl + (trade.pnl ?? 0)).toFixed(2)
   587→    this.dayPnl = +(this.dayPnl + (trade.pnl ?? 0)).toFixed(2)
   588→    if (trade.result === 'WIN') this.stats.wins++
   589→    else if (trade.result === 'LOSS') this.stats.losses++
   590→    this.stats.winRate = this.stats.wins + this.stats.losses > 0
   591→      ? Math.round((this.stats.wins / (this.stats.wins + this.stats.losses)) * 100)
   592→      : 0
   593→
   594→    // deep stats: best/worst trade, streaks, gross pnl (profit factor)
   595→    const pnl = trade.pnl ?? 0
   596→    if (pnl > this.stats.bestTrade) this.stats.bestTrade = pnl
   597→    if (pnl < this.stats.worstTrade) this.stats.worstTrade = pnl
   598→    if (pnl > 0) {
   599→      this.stats.grossProfit = +(this.stats.grossProfit + pnl).toFixed(2)
   600→      this.stats.streak = this.stats.streak > 0 ? this.stats.streak + 1 : 1
   601→      if (this.stats.streak > this.stats.bestStreak) this.stats.bestStreak = this.stats.streak
   602→    } else if (pnl < 0) {
   603→      this.stats.grossLoss = +(this.stats.grossLoss + pnl).toFixed(2)
   604→      this.stats.streak = this.stats.streak < 0 ? this.stats.streak - 1 : -1
   605→      if (this.stats.streak < this.stats.worstStreak) this.stats.worstStreak = this.stats.streak
   606→    }
   607→
   608→    // timestamps for the post-loss cooldown
   609→    if (trade.result === 'LOSS') this.lastLossAt = Date.now()
   610→
   611→    // circuit breaker: pause after N consecutive losses
   612→    const limit = this.settings.maxConsecutiveLosses
   613→    if (trade.result === 'LOSS' && limit > 0 && this.stats.streak <= -limit) {
   614→      this.pausedReason = `${limit} باخت متوالی — بات برای محافظت از سرمایه متوقف شد`
   615→      this.settings.botRunning = false
   616→      this.log('warn', `${this.pausedReason} — پس از استراحت می‌توانید دوباره روشن کنید`)
   617→    }
   618→
   619→    this.save()
   620→  }

   622→  private mt5Refreshing = false
   623→  private async mt5Refresh() {
   624→    if (this.settings.mode !== 'mt5' || this.mt5Refreshing) return
   625→    if (!this.settings.metaapiToken || !this.settings.metaapiAccountId) {
   626→      this.mt5Status = 'error'
   627→      this.mt5Error = 'توکن یا شناسه حساب MetaApi وارد نشده — در تنظیمات تکمیل کنید'
   628→      return
   629→    }
   630→    this.mt5Refreshing = true
   631→    try {
   632→      const res = await mt5Account(this.settings)
   633→      if (res.ok && res.info) {
   634→        this.mt5 = res.info
   635→        this.mt5Status = 'connected'
   636→        this.mt5Error = null
   637→        this.balance = res.info.balance
   638→        if (!this.mt5ConnectedLogged) {
   639→          this.log('success', `اتصال MT5 برقرار شد — ${res.info.server} (${res.info.broker})`)
   640→          this.mt5ConnectedLogged = true
   641→        }
   642→      } else {
   643→        this.mt5Status = 'error'
   644→        this.mt5Error = res.error || 'اتصال ناموفق'
   645→      }
   646→      // update live PnL of open MT5 trades
   647→      if (this.mt5Status === 'connected') {
   648→        const pos = await mt5Positions(this.settings)
   649→        if (pos.ok && Array.isArray(pos.positions)) {
   650→          for (const t of this.trades.filter(t => t.status === 'open' && t.mode === 'mt5')) {
   651→            const p: any = pos.positions.find((x: any) => String(x.id) === t.positionId)
   652→            if (p) t.pnl = +Number(p.profit ?? 0).toFixed(2)
   653→          }
   654→        }
   655→      }
   656→    } finally {
   657→      this.mt5Refreshing = false
   658→      this.onEmit()
   659→    }
   660→  }

   666→    const floating = openTrades.reduce((s, t) => s + (t.pnl && t.status === 'open' ? t.pnl : 0), 0)
   667→    return {
