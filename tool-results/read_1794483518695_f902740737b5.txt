   355→   355→    const now = Date.now()

   357→   357→      if (trade.mode === 'demo' && now >= trade.closeTime) {
   358→   358→        this.settleDemo(trade, this.priceOf(trade.symbol))
   359→   359→      } else if (trade.mode === 'mt5' && trade.positionId && now >= trade.closeTime) {
   360→   360→        this.settleMt5(trade, false)
   361→   361→      }

   367→   367→  private async aiCycle() {
   368→   368→    if (this.aiBusy) return
   369→   369→    // technical analysis always runs at full turbo cadence; only the GLM call
   370→   370→    // is throttled separately by its own rate-limit backoff
   371→   371→    if (Date.now() - this.lastSignalAt < this.settings.signalIntervalSec * 1000) return
   372→   372→    this.aiBusy = true
   373→   373→    try {
   374→   374→      const activeSym = this.settings.activeSymbol
   375→   375→
   376→   376→      // 1) GLM deep analysis of the active symbol (rate-limit aware)
   377→   377→      let glmRan = false
   378→   378→      const glmReady = Date.now() - this.lastGlmTryAt >= this.rateBackoff * 1000
   379→   379→      if (glmReady) {
   380→   380→        this.lastGlmTryAt = Date.now()
   381→   381→        try {
   382→   382→          const candles = this.markets[activeSym].b1m.closed.slice(-40)
   383→   383→          const verdict = await analyzeMarket(candles, activeSym, this.recentSummary())
   384→   384→          this.rateBackoff = 0
   385→   385→          // If GLM hesitates (HOLD) but technicals clearly lean one way, follow the technical signal
   386→   386→          let final = verdict
   387→   387→          if (verdict.action === 'HOLD') {
   388→   388→            const local = localVerdict(candles)
   389→   389→            if (local.action !== 'HOLD' && local.confidence >= 60) {
   390→   390→              final = { ...local, reason: `${local.reason} — تأیید تکنیکال` }
   391→   391→            }
   392→   392→          }
   393→   393→          this.applyVerdict(final, 'AI', activeSym)
   394→   394→          glmRan = true
   395→   395→        } catch (e: any) {
   396→   396→          const msg = String(e?.message || 'نامشخص')
   397→   397→          const rateLimited = msg.includes('429') || msg.toLowerCase().includes('too many')
   398→   398→          if (rateLimited) {
   399→   399→            // exponential backoff up to 120s — for the GLM call only
   400→   400→            this.rateBackoff = Math.min((this.rateBackoff || 15) * 1.6, 120)
   401→   401→          }
   402→   402→          this.log('info', `AI در دسترس نیست (${rateLimited ? 'محدودیت نرخ' : 'خطا'}) — ادامه با تحلیل تکنیکال`)
   403→   403→        }
   404→   404→      }
   405→   405→
   406→   406→      // 2) technical scan of the other symbols so the bot never misses opportunities
   407→   407→      for (const sym of SYMBOLS) {
   408→   408→        if (sym === activeSym && glmRan) continue
   409→   409→        try {
   410→   410→          const candles = this.markets[sym].b1m.closed.slice(-40)
   411→   411→          const verdict = localVerdict(candles, this.perSymbolAccuracy(sym))
   412→   412→          this.applyVerdict(verdict, 'تکنیکال', sym)
   413→   413→        } catch (e: any) {
   414→   414→          this.log('error', `خطای تحلیل تکنیکال ${symbolConfig(sym).fa}: ${e?.message || 'نامشخص'}`)
   415→   415→        }
   416→   416→      }
   417→   417→    } finally {
   418→   418→      this.aiBusy = false
   419→   419→      this.lastSignalAt = Date.now()
   420→   420→      this.onEmit()
   421→   421→    }
   422→   422→  }

   424→   424→  private applyVerdict(
   425→   425→    verdict: { action: 'BUY' | 'SELL' | 'HOLD'; confidence: number; reason: string; indicators: ReturnType<typeof computeIndicators> },
   426→   426→    source: 'AI' | 'تکنیکال',
   427→   427→    symbol: string,
   428→   428→  ) {
   429→   429→    const cfg = symbolConfig(symbol)
   430→   430→    const price = this.priceOf(symbol)
   431→   431→
   432→   432→    let acted = false
   433→   433→    let actedTrade: Trade | null = null
   434→   434→    let blockedReason: string | undefined
   435→   435→    if (verdict.action !== 'HOLD' && verdict.confidence >= this.settings.minConfidence && this.settings.botRunning) {
   436→   436→      const r = this.tryOpenTrade(verdict.action as 'BUY' | 'SELL', verdict.confidence, verdict.reason, undefined, false, symbol)
   437→   437→      acted = r.ok
   438→   438→      actedTrade = r.trade ?? null
   439→   439→      blockedReason = r.reason
   440→   440→    }
   441→   441→
   442→   442→    const signal: AISignal = {
   443→   443→      id: Math.random().toString(36).slice(2),
   444→   444→      time: Date.now(),
   445→   445→      symbol,
   446→   446→      action: verdict.action,
   447→   447→      confidence: verdict.confidence,
   448→   448→      reason: verdict.reason,
   449→   449→      price,
   450→   450→      indicators: verdict.indicators,
   451→   451→      acted,
   452→   452→    }
   453→   453→    // keep signal history: prefer mixing symbols but recent-first ordering stays
   454→   454→    this.signals.unshift(signal)
   455→   455→    if (this.signals.length > 30) this.signals.pop()
   456→   456→
   457→   457→    const label = source === 'AI' ? 'AI' : 'تحلیل تکنیکال (AI موقتاً در دسترس نیست)'
   458→   458→    const tag = verdict.action === 'HOLD' ? '' : acted
   459→   459→      ? ' → معامله باز شد'
   460→   460→      : verdict.confidence >= this.settings.minConfidence
   461→   461→        ? ` (${blockedReason || 'بات غیرفعال یا ظرفیت تکمیل'})`
   462→   462→        : ` (زیر حد اطمینان ${this.settings.minConfidence}%)`
   463→   463→    this.log(source === 'AI' ? 'ai' : 'info', `${label} [${cfg.fa}]: ${verdict.action === 'BUY' ? 'خرید' : verdict.action === 'SELL' ? 'فروش' : 'انتظار'} — اطمینان ${verdict.confidence}٪ — ${verdict.reason}${tag}`)
   464→   464→    if (actedTrade) this.log('success', `معامله ${verdict.action === 'BUY' ? 'خرید' : 'فروش'} ${cfg.fa} با حجم $${actedTrade.stake} باز شد — هدف: ${actedTrade.durationSec} ثانیه`)
   465→   465→  }
   466→   466→
   467→   467→  private canOpen(includeCooldown = false): { ok: boolean; reason?: string } {
   468→   468→    if (this.pausedReason) return { ok: false, reason: this.pausedReason }
   469→   469→    if (this.dayPnl <= -this.settings.dailyLossCap) {
   470→   470→      this.pausedReason = `سقف ضرر روزانه (${this.settings.dailyLossCap}$) لمس شد — بات متوقف است`
   471→   471→      this.settings.botRunning = false
   472→   472→      this.log('warn', this.pausedReason)
   473→   473→      return { ok: false, reason: this.pausedReason }
   474→   474→    }
   475→   475→    // post-loss cooldown: prevents revenge-trading cascades right after a loss
   476→   476→    if (includeCooldown && this.lastLossAt > 0) {
   477→   477→      const waitMs = this.settings.cooldownAfterLossSec * 1000
   478→   478→      const elapsed = Date.now() - this.lastLossAt
   479→   479→      if (elapsed < waitMs) {
   480→   480→        return { ok: false, reason: `استراحت بعد از باخت — ${Math.ceil((waitMs - elapsed) / 1000)} ثانیه تا معامله بعدی` }
   481→   481→      }
   482→   482→    }
   483→   483→    const openCount = this.trades.filter(t => t.status === 'open').length
   484→   484→    if (openCount >= this.settings.maxConcurrent) return { ok: false, reason: 'ظرفیت معاملات همزمان تکمیل است' }
   485→   485→    if (this.settings.mode === 'demo' && this.balance < this.currentStake()) {
   486→   486→      this.pausedReason = 'موجودی کافی نیست'
   487→   487→      return { ok: false, reason: this.pausedReason }
   488→   488→    }
   489→   489→    return { ok: true }
   490→   490→  }

   492→   492→  private tryOpenTrade(direction: 'BUY' | 'SELL', confidence: number, reason: string, comment?: string, skipCooldown = false, symbol = this.settings.activeSymbol): { ok: boolean; reason?: string; trade?: Trade } {
   493→   493→    const check = this.canOpen(!skipCooldown)
   494→   494→    if (!check.ok) return { ok: false, reason: check.reason }
   495→   495→    const now = Date.now()
   496→   496→    const base: Omit<Trade, 'openPrice' | 'mode' | 'positionId' | 'lots' | 'stake'> = {
   497→   497→      id: Math.random().toString(36).slice(2),
   498→   498→      symbol,
   499→   499→      direction,
   500→   500→      openTime: now,
   501→   501→      closeTime: now + this.settings.tradeDurationSec * 1000,
   502→   502→      durationSec: this.settings.tradeDurationSec,
   503→   503→      status: 'open',
   504→   504→      confidence,
   505→   505→      aiReason: reason,
   506→   506→      comment: comment ?? `AI ${confidence}%`,
   507→   507→    }
   508→   508→
   509→   509→    if (this.settings.mode === 'demo') {
   510→   510→      const stake = this.currentStake()
   511→   511→      if (stake > this.balance) return { ok: false, reason: 'موجودی کافی نیست' }
   512→   512→      const trade: Trade = { ...base, mode: 'demo', stake, lots: 0, openPrice: this.priceOf(symbol) }
   513→   513→      this.trades.unshift(trade)
   514→   514→      if (this.trades.length > MAX_HISTORY) this.trades.pop()
   515→   515→      this.save()
   516→   516→      return { ok: true, trade }
   517→   517→    }
   518→   518→
   519→   519→    // MT5 real mode
   520→   520→    const trade: Trade = { ...base, mode: 'mt5', stake: 0, lots: this.settings.mt5Lots, openPrice: this.priceOf(symbol) }
   521→   521→    // optimistic insert; confirm with API
   522→   522→    this.trades.unshift(trade)
   523→   523→    if (this.trades.length > MAX_HISTORY) this.trades.pop()
   524→   524→    mt5Open(this.settings, direction, base.comment, symbol)
   525→   525→      .then(res => {
   526→   526→        if (res.ok) {
   527→   527→          trade.positionId = res.positionId
   528→   528→          trade.openPrice = res.openPrice ?? trade.openPrice
   529→   529→          this.log('success', `سفارش واقعی ${direction} روی MT5 ثبت شد (${symbolConfig(symbol).fa}، حجم ${this.settings.mt5Lots} لات)`)
   530→   530→        } else {
   531→   531→          trade.status = 'closed'
   532→   532→          trade.result = 'DRAW'
   533→   533→          trade.pnl = 0
   534→   534→          this.log('error', res.error || 'ثبت سفارش MT5 ناموفق بود')
   535→   535→        }
   536→   536→        this.save()
   537→   537→        this.onEmit()
   538→   538→      })
   539→   539→    return { ok: true, trade }
   540→   540→  }

   542→   542→  private settleDemo(trade: Trade, closePrice: number) {
   543→   543→    trade.status = 'closed'
   544→   544→    trade.closePrice = +closePrice.toFixed(symbolConfig(trade.symbol).decimals)
   545→   545→    const result = demoSettle(trade.direction, trade.openPrice, trade.closePrice)
   546→   546→    trade.result = result
   547→   547→    trade.pnl = result === 'WIN' ? +(trade.stake * this.settings.payout).toFixed(2)
   548→   548→      : result === 'LOSS' ? -trade.stake : 0
   549→   549→    this.balance = +(this.balance + (trade.pnl ?? 0)).toFixed(2)
   550→   550→    this.applyTradeResult(trade)
   551→   551→    this.log(result === 'WIN' ? 'success' : 'warn',
   552→   552→      `معامله ${result === 'WIN' ? 'برد ✚' : 'باخت ✖'} — ${trade.direction} ${symbolConfig(trade.symbol).fa} — سود/زیان: ${trade.pnl >= 0 ? '+' : ''}$${trade.pnl} (بسته شده در ${trade.durationSec} ثانیه)`)
   553→   553→    if (result === 'LOSS' && this.settings.cooldownAfterLossSec > 0) {
   554→   554→      this.log('info', `استراحت ${this.settings.cooldownAfterLossSec} ثانیه‌ای بعد از باخت فعال شد`)
   555→   555→    }

   558→   558→  private async settleMt5(trade: Trade, manual: boolean) {
   559→   559→    if (!trade.positionId) return
   560→   560→    const res = await mt5Close(this.settings, trade.positionId)
   561→   561→    if (!res.ok) {
   562→   562→      this.log('error', res.error || 'بستن پوزیشن ناموفق بود')
   563→   563→      return
   564→   564→    }
   565→   565→    // fetch actual profit from positions history via account info refresh
   566→   566→    trade.status = 'closed'
   567→   567→    trade.closeTime = Date.now()
   568→   568→    const acct = await mt5Account(this.settings)
   569→   569→    const pnl = typeof res.pnl === 'number' ? res.pnl : 0
   570→   570→    trade.pnl = +pnl.toFixed(2)
   571→   571→    trade.result = pnl > 0 ? 'WIN' : pnl < 0 ? 'LOSS' : 'DRAW'
   572→   572→    if (acct.ok && acct.info) {
   573→   573→      this.mt5 = acct.info
   574→   574→      this.balance = acct.info.balance
   575→   575→    } else {
   576→   576→      this.balance = +(this.balance + pnl).toFixed(2)
   577→   577→    }
   578→   578→    this.applyTradeResult(trade)
   579→   579→    this.log(pnl >= 0 ? 'success' : 'warn', `پوزیشن MT5 ${manual ? 'دستی ' : ''}بسته شد — سود/زیان: ${pnl >= 0 ? '+' : ''}$${trade.pnl}`)
   580→   580→  }

   582→   582→  private applyTradeResult(trade: Trade) {
   583→   583→    this.stats.totalTrades++
   584→   584→    this.stats.totalPnl = +(this.stats.totalPnl + (trade.pnl ?? 0)).toFixed(2)
   585→   585→    this.stats.todayTrades++
   586→   586→    this.stats.todayPnl = +(this.stats.todayPnl + (trade.pnl ?? 0)).toFixed(2)
   587→   587→    this.dayPnl = +(this.dayPnl + (trade.pnl ?? 0)).toFixed(2)
   588→   588→    if (trade.result === 'WIN') this.stats.wins++
   589→   589→    else if (trade.result === 'LOSS') this.stats.losses++
   590→   590→    this.stats.winRate = this.stats.wins + this.stats.losses > 0
   591→   591→      ? Math.round((this.stats.wins / (this.stats.wins + this.stats.losses)) * 100)
   592→   592→      : 0
   593→   593→
   594→   594→    // deep stats: best/worst trade, streaks, gross pnl (profit factor)
   595→   595→    const pnl = trade.pnl ?? 0
   596→   596→    if (pnl > this.stats.bestTrade) this.stats.bestTrade = pnl
   597→   597→    if (pnl < this.stats.worstTrade) this.stats.worstTrade = pnl
   598→   598→    if (pnl > 0) {
   599→   599→      this.stats.grossProfit = +(this.stats.grossProfit + pnl).toFixed(2)
   600→   600→      this.stats.streak = this.stats.streak > 0 ? this.stats.streak + 1 : 1
   601→   601→      if (this.stats.streak > this.stats.bestStreak) this.stats.bestStreak = this.stats.streak
   602→   602→    } else if (pnl < 0) {
   603→   603→      this.stats.grossLoss = +(this.stats.grossLoss + pnl).toFixed(2)
   604→   604→      this.stats.streak = this.stats.streak < 0 ? this.stats.streak - 1 : -1
   605→   605→      if (this.stats.streak < this.stats.worstStreak) this.stats.worstStreak = this.stats.streak
   606→   606→    }
   607→   607→
   608→   608→    // timestamps for the post-loss cooldown
   609→   609→    if (trade.result === 'LOSS') this.lastLossAt = Date.now()
   610→   610→
   611→   611→    // circuit breaker: pause after N consecutive losses
   612→   612→    const limit = this.settings.maxConsecutiveLosses
   613→   613→    if (trade.result === 'LOSS' && limit > 0 && this.stats.streak <= -limit) {
   614→   614→      this.pausedReason = `${limit} باخت متوالی — بات برای محافظت از سرمایه متوقف شد`
   615→   615→      this.settings.botRunning = false
   616→   616→      this.log('warn', `${this.pausedReason} — پس از استراحت می‌توانید دوباره روشن کنید`)
   617→   617→    }
   618→   618→
   619→   619→    this.save()
   620→   620→  }

   622→   622→  private mt5Refreshing = false
   623→   623→  private async mt5Refresh() {
   624→   624→    if (this.settings.mode !== 'mt5' || this.mt5Refreshing) return
   625→   625→    if (!this.settings.metaapiToken || !this.settings.metaapiAccountId) {
   626→   626→      this.mt5Status = 'error'
   627→   627→      this.mt5Error = 'توکن یا شناسه حساب MetaApi وارد نشده — در تنظیمات تکمیل کنید'
   628→   628→      return
   629→   629→    }
   630→   630→    this.mt5Refreshing = true
   631→   631→    try {
   632→   632→      const res = await mt5Account(this.settings)
   633→   633→      if (res.ok && res.info) {
   634→   634→        this.mt5 = res.info
   635→   635→        this.mt5Status = 'connected'
   636→   636→        this.mt5Error = null
   637→   637→        this.balance = res.info.balance
   638→   638→        if (!this.mt5ConnectedLogged) {
   639→   639→          this.log('success', `اتصال MT5 برقرار شد — ${res.info.server} (${res.info.broker})`)
   640→   640→          this.mt5ConnectedLogged = true
   641→   641→        }
   642→   642→      } else {
   643→   643→        this.mt5Status = 'error'
   644→   644→        this.mt5Error = res.error || 'اتصال ناموفق'
   645→   645→      }
   646→   646→      // update live PnL of open MT5 trades
   647→   647→      if (this.mt5Status === 'connected') {
   648→   648→        const pos = await mt5Positions(this.settings)
   649→   649→        if (pos.ok && Array.isArray(pos.positions)) {
   650→   650→          for (const t of this.trades.filter(t => t.status === 'open' && t.mode === 'mt5')) {
   651→   651→            const p: any = pos.positions.find((x: any) => String(x.id) === t.positionId)
   652→   652→            if (p) t.pnl = +Number(p.profit ?? 0).toFixed(2)
   653→   653→          }
   654→   654→        }
   655→   655→      }
   656→   656→    } finally {
   657→   657→      this.mt5Refreshing = false
   658→   658→      this.onEmit()
   659→   659→    }
   660→   660→  }
